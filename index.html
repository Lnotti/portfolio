	if tile.stat == "break":

		generate(1)
		tiles_list.remove(tile)

		world.set_extents(vec(0,0), vec(100,400))
		return True
	if tile.power == "none":
		pass
	if tile.power == "spring":
		player.v.y = 15
	if tile.power == "hat":
		player.has_hat = True
		player.hat_timer = 50
		player.v.y = 10
	return False

def move_tile(obj):
  if obj.stat != "hole":
    if obj.pos.x+(obj.size.x/2) > 200:
      obj.v.x *= -1
    elif obj.pos.x-(obj.size.x/2) <-200:
      obj.v.x *= -1
    obj.pos += obj.v
